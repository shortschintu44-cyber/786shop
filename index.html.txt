<script>
"use strict";

const STORAGE_KEY = "786cm_products_v1";

const DEFAULT_PRODUCTS = [
  { id: "broiler", name: "Broiler Chicken", price: 260, available: true },
  { id: "skinless", name: "Broiler Skinless", price: 280, available: true },
  { id: "broiler-live", name: "Broiler Live", price: 136, available: true },
  { id: "layer", name: "Layer Chicken", price: 230, available: true },
  { id: "farm-live", name: "Farm Live", price: 135, available: true }
];

let products = loadProducts();

document.getElementById("today").textContent =
  "Date: " + new Date().toLocaleDateString("en-IN", {
    day: "2-digit", month: "long", year: "numeric"
  });

function loadProducts() {
  try {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (!saved) return structuredClone(DEFAULT_PRODUCTS);

    const parsed = JSON.parse(saved);
    if (!Array.isArray(parsed)) throw new Error("Invalid data");

    return DEFAULT_PRODUCTS.map(defaultProduct => {
      const savedProduct = parsed.find(p => p.id === defaultProduct.id);
      if (!savedProduct) return { ...defaultProduct };

      return {
        id: defaultProduct.id,
        name: typeof savedProduct.name === "string"
          ? savedProduct.name : defaultProduct.name,
        price: Number.isFinite(Number(savedProduct.price)) &&
          Number(savedProduct.price) >= 0
          ? Number(savedProduct.price) : defaultProduct.price,
        available: typeof savedProduct.available === "boolean"
          ? savedProduct.available : defaultProduct.available
      };
    });
  } catch {
    return structuredClone(DEFAULT_PRODUCTS);
  }
}

function renderProducts() {
  const tbody = document.getElementById("productRows");
  tbody.replaceChildren();

  products.forEach((product, index) => {
    const tr = document.createElement("tr");

    const nameCell = document.createElement("td");
    nameCell.dataset.label = "Product Name";
    const nameInput = document.createElement("input");
    nameInput.type = "text";
    nameInput.value = product.name;
    nameInput.maxLength = 80;
    nameInput.setAttribute("aria-label", "Product name");
    nameInput.addEventListener("input", () => {
      products[index].name = nameInput.value;
    });
    nameCell.appendChild(nameInput);

    const priceCell = document.createElement("td");
    priceCell.dataset.label = "Price (₹/kg)";
    const priceInput = document.createElement("input");
    priceInput.type = "number";
    priceInput.min = "0";
    priceInput.max = "100000";
    priceInput.step = "0.01";
    priceInput.value = product.price;
    priceInput.required = true;
    priceInput.setAttribute("aria-label", "Price in rupees per kg");
    priceInput.addEventListener("input", () => {
      products[index].price = priceInput.value;
    });
    priceCell.appendChild(priceInput);

    const availabilityCell = document.createElement("td");
    availabilityCell.dataset.label = "Available";
    const checkbox = document.createElement("input");
    checkbox.type = "checkbox";
    checkbox.className = "availability";
    checkbox.checked = product.available;
    checkbox.setAttribute("aria-label", "Product available");
    checkbox.addEventListener("change", () => {
      products[index].available = checkbox.checked;
    });
    availabilityCell.appendChild(checkbox);

    tr.append(nameCell, priceCell, availabilityCell);
    tbody.appendChild(tr);
  });
}

function saveProducts() {
  const rows = [...document.querySelectorAll("#productRows tr")];

  for (let i = 0; i < rows.length; i++) {
    const name = rows[i].querySelector('input[type="text"]').value.trim();
    const priceInput = rows[i].querySelector('input[type="number"]');
    const price = Number(priceInput.value);

    if (!name) {
      showStatus("Please enter a name for every product.", true);
      return;
    }
    if (priceInput.value === "" || !Number.isFinite(price) ||
        price < 0 || price > 100000) {
      showStatus("Enter a valid price between ₹0 and ₹100,000.", true);
      priceInput.focus();
      return;
    }

    products[i].name = name;
    products[i].price = price;
  }

  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(products));
    showStatus("Daily rates saved successfully.");
  } catch {
    showStatus("Could not save. Check browser storage settings.", true);
  }
}

function showStatus(message, isError = false) {
  const el = document.getElementById("status");
  el.textContent = message;
  el.style.color = isError ? "#b51220" : "#18733b";
}

document.getElementById("saveBtn").addEventListener("click", saveProducts);

document.getElementById("resetBtn").addEventListener("click", () => {
  if (!confirm("Restore all five default product names and rates?")) return;
  products = structuredClone(DEFAULT_PRODUCTS);
  renderProducts();
  saveProducts();
});

document.getElementById("exportBtn").addEventListener("click", () => {
  const blob = new Blob([JSON.stringify(products, null, 2)], {
    type: "application/json"
  });
  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");
  link.href = url;
  link.download = "786-chicken-mart-rates.json";
  link.click();
  URL.revokeObjectURL(url);
});

document.getElementById("importBtn").addEventListener("click", () => {
  document.getElementById("importFile").click();
});

document.getElementById("importFile").addEventListener("change", async event => {
  const file = event.target.files[0];
  if (!file) return;

  try {
    const imported = JSON.parse(await file.text());
    if (!Array.isArray(imported)) throw new Error("Invalid file");

    const validated = DEFAULT_PRODUCTS.map(defaultProduct => {
      const item = imported.find(p => p.id === defaultProduct.id);
      if (!item) return { ...defaultProduct };
      if (typeof item.name !== "string" ||
          !Number.isFinite(Number(item.price)) ||
          Number(item.price) < 0 || Number(item.price) > 100000 ||
          typeof item.available !== "boolean") {
        throw new Error("Invalid product data");
      }
      return {
        id: defaultProduct.id,
        name: item.name,
        price: Number(item.price),
        available: item.available
      };
    });

    products = validated;
    renderProducts();
    saveProducts();
  } catch {
    showStatus("Invalid backup file. No rates were imported.", true);
  } finally {
    event.target.value = "";
  }
});

renderProducts();
</script>
