function getDailyRates() {
  const saved = localStorage.getItem("786cm_products_v1");
  if (!saved) return null;

  try {
    return JSON.parse(saved);
  } catch {
    return null;
  }
}

// Example: display current rates
const dailyRates = getDailyRates();

if (dailyRates) {
  dailyRates
    .filter(product => product.available)
    .forEach(product => {
      console.log(
        product.name,
        "₹" + product.price + "/kg"
      );
    });
}
